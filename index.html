🛡️ Diagnóstico de Bugs do Sistema de Acesso Corrigidos

✅ CORRIGIDO

Recursão Infinita nas Políticas RLS

Problema: Políticas RLS que chamavam a tabela profiles diretamente no Supabase causavam loop infinito de queries e erro HTTP 500.

Solução: Implementada função is_admin_or_superadmin() com SECURITY DEFINER.

✅ CORRIGIDO

Elevação Indevida de Privilégios no Cadastro

Problema: Usuários podiam enviar flags administrativas pelo payload público do formulário de cadastro.

Solução: Validação atômica e higienização via trigger PostgreSQL handle_new_user().

✅ CORRIGIDO

Ausência de Flags Granulares Individuais

Problema: Privilégios ficavam engessados em cargos, sem permitir permissões pontuais por usuário.

Solução: Adicionadas 7 flags booleanas por usuário + campo custom_flags (JSONB).

✅ CORRIGIDO

Falta de Trilha de Auditoria e Rastreabilidade

Problema: Alterações de cargo ou permissões não geravam histórico de quem executou a alteração.

Solução: Tabela audit_logs integrada com triggers automáticos e gravação de IP.

Script de Migração & Correção do Supabase

Resolve problemas de recursão infinita no RLS, triggers de signup e tabela de profiles.

-- ==============================================================================
-- SUPABASE ACCESS CONTROL (RBAC) & USER CREATION FLAGS SCHEMA
-- Idempotent Migration Script: Fixes all access control bugs and schema issues
-- ==============================================================================

-- 1. Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 2. Create Departments table
CREATE TABLE IF NOT EXISTS public.departments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL UNIQUE,
    code VARCHAR(20) NOT NULL UNIQUE,
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create Profiles Table (Linked to auth.users) with User Creation Flags
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email VARCHAR(255) NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'viewer',
    status VARCHAR(50) NOT NULL DEFAULT 'active',
    department_id UUID REFERENCES public.departments(id) ON DELETE SET NULL,
    avatar_url TEXT,
    phone VARCHAR(30),
    
    -- User Creation & Access Flags (Granular Permissions)
    can_create_users BOOLEAN NOT NULL DEFAULT FALSE,
    can_edit_users BOOLEAN NOT NULL DEFAULT FALSE,
    can_delete_users BOOLEAN NOT NULL DEFAULT FALSE,
    can_manage_roles BOOLEAN NOT NULL DEFAULT FALSE,
    can_view_audit_logs BOOLEAN NOT NULL DEFAULT FALSE,
    can_export_reports BOOLEAN NOT NULL DEFAULT FALSE,
    can_manage_system_settings BOOLEAN NOT NULL DEFAULT FALSE,
    require_password_change BOOLEAN NOT NULL DEFAULT FALSE,
    custom_flags JSONB NOT NULL DEFAULT '{}'::jsonb,
    
    created_by UUID REFERENCES auth.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 4. Create System Audit Logs Table
CREATE TABLE IF NOT EXISTS public.audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    actor_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
    actor_email VARCHAR(255),
    action VARCHAR(100) NOT NULL,
    resource_type VARCHAR(100) NOT NULL,
    resource_id VARCHAR(255),
    details JSONB NOT NULL DEFAULT '{}'::jsonb,
    ip_address VARCHAR(45),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 5. Helper Security Functions (SECURITY DEFINER to avoid RLS Recursion Bugs)
CREATE OR REPLACE FUNCTION public.is_admin_or_superadmin()
RETURNS BOOLEAN LANGUAGE sql SECURITY DEFINER SET search_path = public, auth STABLE AS $$
    SELECT EXISTS (
        SELECT 1 FROM public.profiles 
        WHERE id = auth.uid() 
          AND role IN ('superadmin', 'admin') 
          AND status = 'active'
    );
$$;

-- 6. Trigger on auth.users (Auto-Profile Creation with Flags)
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth AS $$
DECLARE
    default_role VARCHAR(50) := 'operator';
    is_first_user BOOLEAN := FALSE;
BEGIN
    SELECT NOT EXISTS (SELECT 1 FROM public.profiles) INTO is_first_user;
    IF is_first_user THEN
        default_role := 'superadmin';
    ELSIF NEW.raw_user_meta_data->>'role' IS NOT NULL THEN
        default_role := NEW.raw_user_meta_data->>'role';
    END IF;

    INSERT INTO public.profiles (
        id, email, full_name, role, status,
        can_create_users, can_edit_users, can_delete_users,
        can_manage_roles, can_view_audit_logs, can_export_reports,
        can_manage_system_settings
    ) VALUES (
        NEW.id,
        NEW.email,
        COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email, '@', 1)),
        default_role,
        'active',
        COALESCE((NEW.raw_user_meta_data->>'can_create_users')::boolean, (is_first_user OR default_role IN ('admin', 'superadmin', 'manager'))),
        COALESCE((NEW.raw_user_meta_data->>'can_edit_users')::boolean, (is_first_user OR default_role IN ('admin', 'superadmin', 'manager'))),
        COALESCE((NEW.raw_user_meta_data->>'can_delete_users')::boolean, (is_first_user OR default_role = 'superadmin')),
        COALESCE((NEW.raw_user_meta_data->>'can_manage_roles')::boolean, (is_first_user OR default_role IN ('admin', 'superadmin'))),
        COALESCE((NEW.raw_user_meta_data->>'can_view_audit_logs')::boolean, (is_first_user OR default_role IN ('admin', 'superadmin', 'manager'))),
        COALESCE((NEW.raw_user_meta_data->>'can_export_reports')::boolean, (is_first_user OR default_role IN ('admin', 'superadmin', 'manager', 'supervisor'))),
        COALESCE((NEW.raw_user_meta_data->>'can_manage_system_settings')::boolean, (is_first_user OR default_role = 'superadmin'))
    );
    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- 7. Enable RLS
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.departments ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.audit_logs ENABLE ROW LEVEL SECURITY;

-- 8. Policies
DROP POLICY IF EXISTS "Profiles read" ON public.profiles;
CREATE POLICY "Profiles read" ON public.profiles FOR SELECT TO authenticated USING (TRUE);

DROP POLICY IF EXISTS "Profiles update" ON public.profiles;
CREATE POLICY "Profiles update" ON public.profiles FOR UPDATE TO authenticated USING (
    public.is_admin_or_superadmin() OR auth.uid() = id
);

Como aplicar no seu painel Supabase:

  1. Acesse o painel do seu projeto no Supabase (supabase.com/dashboard).
  2. No menu lateral esquerdo, selecione SQL Editor.
  3. Clique em New Query, cole o código acima e clique em Run.
  4. Pronto! Todas as tabelas, RLS, functions e triggers serão atualizados instantaneamente.
